```typescript
  return `${greeting}, ${name}!`;

// 剩余参数
```

### Client Component
```typescript
// components/user-list.tsx
'use client';

import { useState } from 'react';

export function UserList({ users }: { users: User[] }) {
  const [filter, setFilter] = useState('');
  
  const filtered = users.filter(u => 
    u.name.toLowerCase().includes(filter.toLowerCase())
  );
  
  return (
    <div>
      <input 
        value={filter}
        onChange={(e) => setFilter(e.target.value)}
        placeholder="Filter users..."
      />
      <ul>
        {filtered.map(user => (
          <li key={user.id}>{user.name}</li>
        ))}
      </ul>
    </div>
  );
}
```

```typescript
// actions/users.ts

import { db } from '@/db';
import { revalidatePath } from 'next/cache';

export async function createUser(formData: FormData) {
  const name = formData.get('name') as string;
  const email = formData.get('email') as string;

    </form>
  );
}
```

## 10. 常见错误和解决方案

### 错误 1: Hydration mismatch
```
Error: Text content does not match server-rendered HTML
```

**原因**：Server Component 和 Client Component 渲染结果不一致

**解决**：
```typescript
// ❌ 错误：服务端和客户端时间不同
export default function Page() {
  return <div>{new Date().toISOString()}</div>;
}

// ✅ 正确：使用 Client Component + useEffect
'use client';
export default function Page() {
  const [time, setTime] = useState<string | null>(null);
  
  useEffect(() => {
    setTime(new Date().toISOString());
  }, []);
  
  return <div>{time || 'Loading...'}</div>;
}
```

### 错误 2: useState/useEffect 在 Server Component
```
Error: useState only works in Client Components
```

**解决**：添加 `'use client'` 指令

```typescript
'use client';  // 必须在文件顶部

import { useState } from 'react';
```

### 错误 3: 忘记 key prop
```
Warning: Each child in a list should have a unique "key" prop
```

**解决**：
```typescript
{users.map(user => (
  <UserCard key={user.id} user={user} />
))}
```

### 错误 4: 异步操作未处理错误
```typescript
// ❌ 错误：未处理错误
async function fetchData() {
  const response = await fetch('/api/data');
  const data = await response.json();
  return data;
}

// ✅ 正确：处理错误
async function fetchData() {
  try {
    const response = await fetch('/api/data');
    if (!response.ok) {
      throw new Error('Failed to fetch');
    }
    const data = await response.json();
    return data;
  } catch (error) {
    console.error('Fetch error:', error);
    throw error;
  }
}
```

## 12. Git 工作流（项目规范）

### Commit Message 格式
```bash
<type>(<scope>): <subject>

# 示例
feat(creator): 添加达人卡片组件
fix(ui): 修复按钮样式问题
docs(readme): 更新安装说明
```

### Type 类型
- `feat` - 新功能
- `fix` - Bug 修复
- `docs` - 文档
- `style` - 代码格式
- `refactor` - 重构
- `test` - 测试
- `chore` - 构建/工具

### 提交前检查
```bash
# 1. 代码格式化
pnpm lint

# 2. 类型检查
pnpm typecheck

# 3. 查看改动
git diff

# 4. 分阶段提交
git add src/components/UserCard.tsx
git commit -m "feat(ui): 添加用户卡片组件"
```
